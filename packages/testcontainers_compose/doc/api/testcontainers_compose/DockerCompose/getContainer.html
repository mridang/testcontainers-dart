ComposeContainer getContainer([
  1. String? serviceName,
  2. bool includeAll = false
])

Returns a single ComposeContainer from the stack.

Parameters:

  • serviceName — the Compose service name to look up. When null, there must be exactly one container running in the stack.
  • includeAll — include stopped containers in the search.

Throws ContainerIsNotRunning when:

  • serviceName is null and the stack does not have exactly one container.
  • serviceName is given but no matching container is found.

Implementation

ComposeContainer getContainer([
  String? serviceName,
  bool includeAll = false,
]) {
  if (serviceName == null) {
    final containers = getContainers(includeAll: includeAll);
    return _getOnlyElementOrRaise(
      containers,
      () => ContainerIsNotRunning(
        'get_container failed because no service_name given '
        'and there is not exactly 1 container (but ${containers.length})',
      ),
    );
  }

  final matching = getContainers(includeAll: includeAll)
      .where((c) => c.service == serviceName)
      .toList();

  if (matching.isEmpty) {
    throw ContainerIsNotRunning(
      '$serviceName is not running in the compose context',
    );
  }

  matching.first._dockerCompose = this;
  return matching.first;
}
